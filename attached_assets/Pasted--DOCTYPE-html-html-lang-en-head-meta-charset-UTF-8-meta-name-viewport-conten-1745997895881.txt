    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const bgMusic = document.getElementById('bgMusic');
            const introAudio = document.getElementById('introAudio');
            const toggleBgMusicBtn = document.getElementById('toggleBgMusic');
            const playIntroBtn = document.getElementById('playIntro');
            const musicProgress = document.getElementById('musicProgress');
            const navItems = document.querySelectorAll('.nav-item');
            const sectionContents = document.querySelectorAll('.section-content');
            
            // Initialize with background music paused
            bgMusic.volume = 0.3;
            bgMusic.pause();
            
            // Toggle background music play/pause
            toggleBgMusicBtn.addEventListener('click', function() {
                if (bgMusic.paused) {
                    bgMusic.play();
                    toggleBgMusicBtn.innerHTML = '<i class="fas fa-pause"></i>';
                } else {
                    bgMusic.pause();
                    toggleBgMusicBtn.innerHTML = '<i class="fas fa-play"></i>';
                }
            });
            
            // Play introduction audio
            playIntroBtn.addEventListener('click', function() {
                if (introAudio.paused) {
                    // Pause background music and reduce its volume
                    if (!bgMusic.paused) {
                        bgMusic.volume = 0.1;
                    }
                    
                    introAudio.play();
                    playIntroBtn.innerHTML = '<i class="fas fa-pause mr-2"></i>Pause Introduction';
                    
                    // When intro ends, restore background music volume
                    introAudio.onended = function() {
                        playIntroBtn.innerHTML = '<i class="fas fa-play mr-2"></i>Play Introduction';
                        if (!bgMusic.paused) {
                            bgMusic.volume = 0.3;
                        }
                    };
                } else {
                    introAudio.pause();
                    playIntroBtn.innerHTML = '<i class="fas fa-play mr-2"></i>Play Introduction';
                    
                    // Restore background music volume
                    if (!bgMusic.paused) {
                        bgMusic.volume = 0.3;
                    }
                }
            });
            
            // Update music progress bar
            bgMusic.addEventListener('timeupdate', function() {
                const percentage = (bgMusic.currentTime / bgMusic.duration) * 100;
                musicProgress.style.width = percentage + '%';
            });
            
            // Navigation between sections
            navItems.forEach(item => {
                item.addEventListener('click', function() {
                    const sectionId = this.getAttribute('data-section');
                    
                    // Update active nav item
                    navItems.forEach(nav => nav.classList.remove('active'));
                    this.classList.add('active');
                    
                    // Show selected section
                    sectionContents.forEach(section => {
                        section.classList.remove('active-section');
                        if (section.id === sectionId) {
                            section.classList.add('active-section');
                        }
                    });
                });
            });
            
            // Function to switch sections from playlist cards
            window.switchSection = function(sectionId) {
                // Find and click the nav item
                navItems.forEach(nav => {
                    if (nav.getAttribute('data-section') === sectionId) {
                        nav.click();
                    }
                });
            };
            
            // Optional: Start playing background music automatically after user interaction
            document.body.addEventListener('click', function() {
                if (bgMusic.paused && !bgMusic.played.length) {
                    bgMusic.play();
                    toggleBgMusicBtn.innerHTML = '<i class="fas fa-pause"></i>';
                }
            }, { once: true });
        });
    </script>
